Функция ПолучитьНачальныйОстаток311(Год)

	Счет = ПланыСчетов.Хозрасчетный.НайтиПоКоду("311");

	Организация = Справочники.Организации.НайтиПоНаименованию(
		"ОСББ ""ХЕРСОНСЬКЕ ШОСЕ,38"""
	);

	НачалоПериода = Дата(Год, 1, 1);
	КонецПериода = НачалоПериода;

	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ
	|	ОстаткиИОбороты.СуммаНачальныйОстатокДт КАК НачальныйДт,
	|	ОстаткиИОбороты.СуммаНачальныйОстатокКт КАК НачальныйКт
	|ИЗ
	|	РегистрБухгалтерии.Хозрасчетный.ОстаткиИОбороты(
	|		&НачалоПериода,
	|		&КонецПериода,
	|		Регистратор,
	|		ДвиженияИГраницыПериода,
	|		Счет В (&Счет),
	|		,
	|		Организация = &Организация
	|	) КАК ОстаткиИОбороты";

	Запрос.УстановитьПараметр("НачалоПериода", НачалоПериода);
	Запрос.УстановитьПараметр("КонецПериода", КонецПериода);
	Запрос.УстановитьПараметр("Счет", Счет);
	Запрос.УстановитьПараметр("Организация", Организация);

	Выборка = Запрос.Выполнить().Выбрать();

	НачальныйОстаток = 0;

	Если Выборка.Следующий() Тогда

		Если ЗначениеЗаполнено(Выборка.НачальныйДт) Тогда
			НачальныйОстаток = Выборка.НачальныйДт;
		КонецЕсли;

	КонецЕсли;

	Возврат НачальныйОстаток;

		|	РегистрБухгалтерии.Хозрасчетный.Обороты(

		|		Регистратор,
		|		Счет В (&Счет),

		|		Организация = &Организация

		Запрос.УстановитьПараметр("Счет", Счет);
		Запрос.УстановитьПараметр("Организация", Организация);

		Выборка = Запрос.Выполнить().Выбрать();

		Дебет = 0;

		Если Выборка.Следующий() Тогда
			Если ЗначениеЗаполнено(Выборка.Дебет) Тогда
				Дебет = Выборка.Дебет;

		КонецЕсли;

		СтрокаРезультата.Вставить("Дебет", Дебет);

		Результаты.Добавить(СтрокаРезультата);

			СтрокиМесяца.Добавить(СтрокаРезультата);

		КонецЦикла;

		Результаты.Добавить(СтрокиМесяца);

	КонецЦикла;

	Возврат Результаты;

&НаСервере
Процедура ЗаписатьВExcel(
	Результаты301,
	Результаты311Кт301,
	Результаты311Кт3771,
	Результаты949Кт311,
	Результаты311Кт949,
	НачальныйОстаток311,
	Год
)

	ПутьКФайлу = "D:\баланс.xlsx";

	Excel = Новый COMОбъект("Excel.Application");

	Excel.Visible = Истина;

	Книга = Excel.Workbooks.Open(ПутьКФайлу);

	Лист = Книга.Worksheets("отчет");

	// ==========================================
	// Заголовки отчета
	// ==========================================

	ЗаписатьЗаголовкиВExcel(
		Лист,
		Год
	);

	// ==========================================
	// Начальный остаток 311
	// ==========================================

	ЗаписатьНачальныйОстатокВExcel(
		Лист,
		НачальныйОстаток311
	);

	// ==========================================
	// Очищаем старые детальные данные
	//
	// Дт 311 -> Кт 949
	// и подробные расходы 949 -> 311
	//
	// Строка 20 НЕ ТРОГАЕТСЯ
	// ==========================================

	Лист.Range("A64:N1000").ClearContents();


	// ==========================================
	// Формируем список уникальных аналитик
	//
	// Дт 311 -> Кт 949
	// ==========================================

	УникальныеАналитики =
		Новый Массив;


	Для НомерМесяца = 1 По 12 Цикл

		СтрокиМесяца =
			Результаты311Кт949[
				НомерМесяца - 1
			];


		Для Каждого СтрокаРезультата
			Из СтрокиМесяца Цикл

			ИмяАналитики =
				СтрокаРезультата.Аналитика;


			Найдена = Ложь;


			Для Каждого СуществующаяАналитика
				Из УникальныеАналитики Цикл

				Если СуществующаяАналитика =
					ИмяАналитики Тогда

					Найдена = Истина;

					Прервать;

				КонецЕсли;

			КонецЦикла;


			Если НЕ Найдена Тогда

				УникальныеАналитики.Добавить(
					ИмяАналитики
				);

			КонецЕсли;

		КонецЦикла;

	КонецЦикла;


	// ==========================================
	// Записываем детальные аналитики
	//
	// Дт 311 -> Кт 949
	// начиная с A64
	// ==========================================

	Для НомерАналитики = 0
		По УникальныеАналитики.Количество() - 1 Цикл

		СтрокаExcel =
			64 + НомерАналитики;


		ИмяАналитики =
			УникальныеАналитики[
				НомерАналитики
			];


		Лист.Cells(
			СтрокаExcel,
			1
		).Value =
			ИмяАналитики;


		Для НомерМесяца = 1 По 12 Цикл

			НомерКолонки =
				НомерМесяца + 2;


			СтрокиМесяца =
				Результаты311Кт949[
					НомерМесяца - 1
				];


			СуммаЗаМесяц = 0;


			Для Каждого СтрокаРезультата
				Из СтрокиМесяца Цикл

				Если СтрокаРезультата.Аналитика =
					ИмяАналитики Тогда

					СуммаЗаМесяц =
						СуммаЗаМесяц
						+ СтрокаРезультата.Сумма;

				КонецЕсли;

			КонецЦикла;


			Если СуммаЗаМесяц <> 0 Тогда

				Лист.Cells(
					СтрокаExcel,
					НомерКолонки
				).Value =
					СуммаЗаМесяц;

			Иначе

				Лист.Cells(
					СтрокаExcel,
					НомерКолонки
				).ClearContents();

			КонецЕсли;

		КонецЦикла;

	КонецЦикла;


	// ==========================================
	// Первая свободная строка после блока
	//
	// Дт 311 -> Кт 949
	// ==========================================

	ПерваяСтрокаПрочихРасходов =
		64 + УникальныеАналитики.Количество();


	Если ПерваяСтрокаПрочихРасходов < 65 Тогда

		ПерваяСтрокаПрочихРасходов =
			65;

	КонецЕсли;


	// ==========================================
	// Формируем словарь всех прочих расходов
	//
	// Ключ =
	// наименование аналитики
	//
	// Значение =
	// массив из 12 месячных сумм
	// ==========================================

	ПрочиеПоНаименованиям =
		Новый Соответствие;


	Для НомерМесяца = 1 По 12 Цикл

		Строка949Кт311 =
			Результаты949Кт311[
				НомерМесяца - 1
			];


		// В Результаты949Кт311
		// поле ПрочиеРасходы содержит
		// аналитику, которая не относится
		// к шести основным статьям.

		Для Каждого СтрокаПрочего
			Из Строка949Кт311.ПрочиеРасходы Цикл

			Наименование =
				СтрокаПрочего.Аналитика;


			Если НЕ ЗначениеЗаполнено(
				Наименование
			) Тогда

				Продолжить;

			КонецЕсли;


			Если ПрочиеПоНаименованиям.Получить(
				Наименование
			) = Неопределено Тогда

				МассивМесяцев =
					Новый Массив;

				Для Индекс = 1 По 12 Цикл

					МассивМесяцев.Добавить(0);

				КонецЦикла;

				ПрочиеПоНаименованиям.Вставить(
					Наименование,
					МассивМесяцев
				);

			КонецЕсли;

			МассивМесяцев =
				ПрочиеПоНаименованиям[
					Наименование
				];


			МассивМесяцев[
				НомерМесяца - 1
			] =
				МассивМесяцев[
					НомерМесяца - 1
				]
				+ СтрокаПрочего.Сумма;

		КонецЦикла;

	КонецЦикла;


	// ==========================================
	// Подробная расшифровка прочих расходов
	//
	// Начиная с первой свободной строки
	// после блока Дт 311 -> Кт 949
	// ==========================================

	СтрокаExcel =
		ПерваяСтрокаПрочихРасходов;


	Для Каждого Элемент
		Из ПрочиеПоНаименованиям Цикл

		Наименование =
			Элемент.Ключ;


		МассивМесяцев =
			Элемент.Значение;


		// Наименование расхода -> колонка A

		Лист.Cells(
			СтрокаExcel,
			1
		).Value =
			Наименование;


		// Январь -> C
		// Февраль -> D
		// ...
		// Декабрь -> N

		Для НомерМесяца = 1 По 12 Цикл

			НомерКолонки =
				НомерМесяца + 2;


			СуммаМесяца =
				МассивМесяцев[
					НомерМесяца - 1
				];


			Если СуммаМесяца <> 0 Тогда

				Лист.Cells(
					СтрокаExcel,
					НомерКолонки
				).Value =
					СуммаМесяца;

			Иначе

				Лист.Cells(
					СтрокаExcel,
					НомерКолонки
				).ClearContents();

			КонецЕсли;

		КонецЦикла;


		СтрокаExcel =
			СтрокаExcel + 1;

	КонецЦикла;


	// ==========================================
	// Основные месячные данные
	// ==========================================

	Для НомерМесяца = 1 По 12 Цикл

		НомерКолонки =
			НомерМесяца + 2;


		// ==========================================
		// Дт 301 -> строки 43 и 59
		// ==========================================

		Строка301 =
			Результаты301[
				НомерМесяца - 1
			];


		Если Строка301.Дебет <> 0 Тогда

			Лист.Cells(
				43,
				НомерКолонки
			).Value =
				Строка301.Дебет;


			Лист.Cells(
				59,
				НомерКолонки
			).Value =
				Строка301.Дебет;

		Иначе

			Лист.Cells(
				43,
				НомерКолонки
			).ClearContents();


			Лист.Cells(
				59,
				НомерКолонки
			).ClearContents();

		КонецЕсли;


		// ==========================================
		// Дт 311 / Кт 301 -> строки 46 и 8
		// ==========================================

		Строка311Кт301 =
			Результаты311Кт301[
				НомерМесяца - 1
			];


		Если Строка311Кт301.Сумма <> 0 Тогда

			Лист.Cells(
				46,
				НомерКолонки
			).Value =
				Строка311Кт301.Сумма;


			Лист.Cells(
				8,
				НомерКолонки
			).Value =
				Строка311Кт301.Сумма;

		Иначе

			Лист.Cells(
				46,
				НомерКолонки
			).ClearContents();


			Лист.Cells(
				8,
				НомерКолонки
			).ClearContents();

		КонецЕсли;


		// ==========================================
		// Дт 311 / Кт 3771 -> строки 7 и 60
		// ==========================================

		Строка311Кт3771 =
			Результаты311Кт3771[
				НомерМесяца - 1
			];


		Если Строка311Кт3771.Сумма <> 0 Тогда

			Лист.Cells(
				7,
				НомерКолонки
			).Value =
				Строка311Кт3771.Сумма;


			Лист.Cells(
				60,
				НомерКолонки
			).Value =
				Строка311Кт3771.Сумма;

		Иначе

			Лист.Cells(
				7,
				НомерКолонки
			).ClearContents();

				НомерКолонки
			).ClearContents();

			Результаты949Кт311[
