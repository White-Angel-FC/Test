//   Блок "р/с"        : заголовок A2, остаток A4/B4
//                       поступления  : C7 :N7   (311 <- 3771)
//                       снятие в кассу: C8 :N8   (311 <- 301)
//                       расходы      : C13:N20  (949 <- 311)
//   Блок "касса"      : заголовок A38, остаток A40
//                       поступления  : C43:N43 (301 Дт)
//                       снятие с р/с : C46:N46 (311 <- 301)
//   Блок "взаєморозр.": заголовок A54, остаток A56
//                       взносы       : C59:N59 (301 Дт)
//                       поступления  : C60:N60 (311 <- 3771)
//   Детальные блоки (A64+):
//       сначала 311 <- 949 (уникальные аналитики),
//       затем 949 <- 311 (уникальные прочие расходы).

	ЗаписатьВExcel(
		ПарамОтчёта,
		Год,
		Результаты301,
		Результаты311Кт301,
		Результаты311Кт3771,
		Результаты949Кт311,
		Результаты311Кт949,
